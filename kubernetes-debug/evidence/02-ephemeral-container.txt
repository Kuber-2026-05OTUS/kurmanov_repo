### Команда создания эфемерного контейнера (kubectl debug):
kubectl debug -it nginx-distroless --image=nicolaka/netshoot:latest --image-pull-policy=IfNotPresent \
  --target=nginx --container=debugger2 --profile=netadmin -- /bin/bash

# Для неинтерактивного снятия доказательств контейнер создан долгоживущим:
kubectl debug nginx-distroless --image=nicolaka/netshoot:latest --image-pull-policy=IfNotPresent \
  --target=nginx --container=debugger2 --profile=netadmin -- sleep infinity
